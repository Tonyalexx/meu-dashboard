    // CSV Upload & Parsing
    function handleCSVUpload(event) {
      const file = event.target.files[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function(e) {
        const text = e.target.result;
        const lines = text.split('\n');
        if (lines.length < 2) return;

        const importedRows = [];
        let cumulativeProfit = 0;
        const curvePoints = [];

        for (let i = 1; i < lines.length; i++) {
          const cols = lines[i].split(',');
          if (cols.length >= 5) {
            const profit = parseFloat(cols[1]) || 0;
            cumulativeProfit += profit;
            curvePoints.push(cumulativeProfit);

            importedRows.push({
              rank: i,
              version: `EA Run #${i}`,
              opt: 'v3.11',
              tf: 'M15',
              hour: '09:00',
              pf: parseFloat(cols[4]) || 1.8,
              profit: profit,
              winRate: (cols[6] || '55') + '%',
              trades: parseInt(cols[7]) || 100,
              sharpe: parseFloat(cols[9]) || 0.8
            });
          }
        }

        if (importedRows.length > 0) {
          tableData = importedRows;
          renderTable(tableData);
          updateEquityChart(curvePoints);
          
          // Update KPIs from last imported row
          document.getElementById('kpi-net-profit').innerText = `$${cumulativeProfit.toLocaleString('en-US', {minimumFractionDigits: 2})}`;
        }
      };
      reader.readAsText(file);
    }

    // Google Sheets Direct Live Sync
    function promptGoogleSheetsSync() {
      const sheetUrl = prompt("Enter your Google Sheet CSV Export URL (or shareable published link):");
      if (!sheetUrl) return;

      fetch(sheetUrl)
        .then(res => res.text())
        .then(csvText => {
          const fakeEvent = { target: { files: [new Blob([csvText], {type: 'text/csv'})] } };
          handleCSVUpload(fakeEvent);
          alert("Successfully synced with Google Sheets!");
        })
        .catch(err => alert("Error syncing Google Sheet: Make sure the link is publicly viewable."));
    }
